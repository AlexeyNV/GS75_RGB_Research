cmake_minimum_required(VERSION 3.22)
project(GS75_RGB_Research C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(GLOB APP CONFIGURE_DEPENDS App/Src/*.c Core/Src/*.c USB_DEVICE/App/*.c USB_DEVICE/Target/*.c)
file(GLOB HAL CONFIGURE_DEPENDS Drivers/STM32F4xx_HAL_Driver/Src/*.c)
set_source_files_properties(${HAL} PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")
set(USB Middlewares/ST/STM32_USB_Device_Library)
add_executable(${PROJECT_NAME} ${APP} ${HAL} ${USB}/Core/Src/usbd_core.c ${USB}/Core/Src/usbd_ctlreq.c ${USB}/Core/Src/usbd_ioreq.c ${USB}/Class/CDC/Src/usbd_cdc.c startup_stm32f411xe.s)
set_target_properties(${PROJECT_NAME} PROPERTIES SUFFIX .elf)
target_include_directories(${PROJECT_NAME} PRIVATE App/Inc Core/Inc USB_DEVICE/App USB_DEVICE/Target Drivers/CMSIS/Include Drivers/CMSIS/Device/ST/STM32F4xx/Include Drivers/STM32F4xx_HAL_Driver/Inc ${USB}/Core/Inc ${USB}/Class/CDC/Inc)
target_compile_definitions(${PROJECT_NAME} PRIVATE STM32F411xE USE_HAL_DRIVER HSE_VALUE=8000000U)
set(CPU -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard)
target_compile_options(${PROJECT_NAME} PRIVATE ${CPU} -Os -g3 -ffunction-sections -fdata-sections -Wall -Wextra -Werror)
target_link_options(${PROJECT_NAME} PRIVATE ${CPU} -T${CMAKE_CURRENT_SOURCE_DIR}/STM32F411RE_FLASH.ld --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--no-warn-rwx-segments -Wl,-Map=${PROJECT_NAME}.map -Wl,--print-memory-usage)
target_link_libraries(${PROJECT_NAME} PRIVATE c m)
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.hex COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${PROJECT_NAME}> ${PROJECT_NAME}.bin COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}>)
