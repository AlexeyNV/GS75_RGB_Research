cmake_minimum_required(VERSION 3.22)
project(rgb_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(rgb_tests test_firmware.c ../firmware/App/Src/mux4051.c ../firmware/App/Src/current_sense.c ../firmware/App/Src/safety.c ../firmware/App/Src/cli.c)
target_include_directories(rgb_tests PRIVATE ../firmware/App/Inc)
if(MSVC)
  target_compile_options(rgb_tests PRIVATE /W4 /WX /UNDEBUG)
  target_compile_definitions(rgb_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(rgb_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME firmware_logic COMMAND rgb_tests)
