cmake_minimum_required(VERSION 3.22)
project(rgb_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(APP_SOURCES ../firmware/App/Src/mux4051.c ../firmware/App/Src/current_sense.c ../firmware/App/Src/safety.c ../firmware/App/Src/cli.c)
add_executable(rgb_tests test_firmware.c adc_mock.c ../firmware/Core/Src/adc.c ${APP_SOURCES})
add_executable(adc_tests test_adc.c adc_mock.c ../firmware/Core/Src/adc.c ${APP_SOURCES})
foreach(target rgb_tests adc_tests)
target_include_directories(${target} PRIVATE stubs ../firmware/App/Inc)
if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
endforeach()
add_test(NAME firmware_logic COMMAND rgb_tests)
add_test(NAME adc_acquisition COMMAND adc_tests)
